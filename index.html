            }

        function sanitizePasscode(raw) {

            return String(raw || '').trim().toLowerCase().replace(/[^a-z0-9_-]/g, '-');

        async function pushCurrentStateToCloud() {

            if (!db || !currentUser || !window.activeSyncPasscode || isApplyingRemoteUpdate) return;

            const code = sanitizePasscode(window.activeSyncPasscode);

            if (!code) return;

            try {

            const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'macro_sync_rooms', code);

            getDoc(docRef).then((snap) => {

            }).catch((err) => {

                console.error('讀取雲端頻道失敗:', err);

            });

                docRef,

                }

            );

        }

                    let h = img.height;

                    if (w > maxDim || h > maxDim) {

                        if (w > h) { h = Math.round((h * maxDim) / w); w = maxDim; }

                        else { w = Math.round((w * maxDim) / h); h = maxDim; }

                    }

                    const canvas = document.createElement('canvas');

                    canvas.width = w;

                    canvas.height = h;

                    const ctx = canvas.getContext('2d');

                    ctx.drawImage(img, 0, 0, w, h);

                    const base64Data = canvas.toDataURL('image/jpeg', 0.82).split(',')[1];


                    const prompt = `請讀取這張 iPhone「健身 (Fitness)」或「健康 (Apple Health)」或 Apple Watch 畫面的截圖，擷取使用者的消耗熱量數據（單位為大卡/kcal/卡）。

請注意以下規則：

1. 若畫面中有顯示「靜息能量 (Resting Energy)」，請填入 restingKcal（若無顯示則填 null）。

請只回傳 JSON 物件，格式：{"restingKcal": 數字或null, "totalActiveKcal": 數字或null, "workouts": [{"name": "運動名稱", "mins": 分鐘數, "kcal": 消耗大卡}]}`;

                        contents: [{

                        const workoutKcalSum = detectedWorkouts.reduce((s, item) => s + (Number(item.kcal) || 0), 0);

                            burnState.dayType = 'workout';

                        renderAll();

        // 多模型自動備援呼叫 (依序嘗試 gemini-3-flash-preview, gemini-2.5-flash, gemini-2.0-flash, gemini-1.5-flash)

        async function callGeminiModelsWithFallback(payload, overrideKey = null) {

            const apiKey = overrideKey !== null ? overrideKey : getActiveApiKey();

            const candidateModels = [

                'gemini-3-flash-preview',

                'gemini-2.5-flash',

                'gemini-2.0-flash',

                'gemini-1.5-flash'

            ];


            let lastError = null;


            for (const modelName of candidateModels) {

                const statusEl = document.getElementById('aiLoadingStatusText');

                if (statusEl) {

                    statusEl.textContent = `AI 正在分析相片中的食材與宏量營養素 (${modelName})...`;

                }


                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`;


                try {

                    const response = await fetch(apiUrl, {

                        method: 'POST',

                        headers: { 'Content-Type': 'application/json' },

                        body: JSON.stringify(payload)

                    });


                    if (response.ok) {

                        const data = await response.json();

                        return { data, modelUsed: modelName };

                    }


                    const errJson = await response.json().catch(() => ({}));

                    const errMsg = errJson?.error?.message || `HTTP ${response.status}`;


                    // 若是 API Key 無效或未提供，直接中斷並提示設定金鑰

                    if (response.status === 400 && (errMsg.toLowerCase().includes('api key') || errMsg.toLowerCase().includes('key not valid'))) {

                        throw new Error('API 金鑰無效或尚未設定，請點擊右上角「設定 API 金鑰」輸入您的 Gemini API Key。');

                    }

                    if (response.status === 403) {

                        throw new Error(`金鑰權限不足或未提供 API Key (${errMsg})。請點擊「設定 API 金鑰」填入有效的 Gemini API Key。`);

                    }


                    lastError = new Error(`[${modelName}] ${errMsg}`);

                } catch (err) {

                    lastError = err;

                    // 若明確為金鑰問題則不繼續嘗試其他模型

                    if (err.message && (err.message.includes('金鑰') || err.message.includes('API Key'))) {

                        throw err;

                    }

                }

            }


            throw lastError || new Error('無法連線至 Gemini 伺服器，請檢查網路或 API 金鑰設定。');

        }


        // 強健的 JSON 陣列解析器（自動去除 Markdown 程式碼區塊或前後說明文字）

        function parseAiJsonResponse(rawText) {

            if (!rawText) throw new Error('AI 回傳內容為空');

            let cleaned = rawText.trim();

            // 移除 ```json ... ``` 標籤

            cleaned = cleaned.replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/i, '').trim();


            try {

                const direct = JSON.parse(cleaned);

                if (Array.isArray(direct)) return direct;

                if (direct && Array.isArray(direct.items)) return direct.items;

                if (direct && typeof direct === 'object') return [direct];

            } catch (e) {

                // 嘗試用正則表達式擷取第一個 [...] 區段

                const arrayMatch = cleaned.match(/\[[\s\S]*\]/);

                if (arrayMatch) {

                    return JSON.parse(arrayMatch[0]);

                }

                throw new Error('無法解析 AI 回傳的營養數據格式');

            }

            return [];

        }

        // 計算當日總消耗熱量 (TDEE) = 靜息能量 + 日常活動紅圈 + 健身運動項目總和

        function calculateTotalBurnKcal() {

            const workoutSum = (burnState.workouts || []).reduce((sum, w) => sum + (Number(w.kcal) || 0), 0);

            const resting = Number(burnState.restingKcal) || 0;

            const activeBase = Number(burnState.activeBaseKcal) || 0;

            return {

                resting,

                activeBase,

                workoutSum,

                totalActiveRedRing: activeBase + workoutSum,

                totalTdee: Math.round(resting + activeBase + workoutSum)

            };

        }

        // 計算單項或總體的 PFC 熱量百分比 (P:4kcal/g, F:9kcal/g, C:4kcal/g)

        function calculatePFCRatios(p, f, c) {

            const pKcal = p * 4;

            const fKcal = f * 9;

            const cKcal = c * 4;

            const totalMacroKcal = pKcal + fKcal + cKcal;


            if (totalMacroKcal <= 0) {

                return { pPct: 0, fPct: 0, cPct: 0, pKcal: 0, fKcal: 0, cKcal: 0, totalMacroKcal: 0 };

            }


            const pPct = Math.round((pKcal / totalMacroKcal) * 100);

            const fPct = Math.round((fKcal / totalMacroKcal) * 100);

            // 確保總和為 100%

            const cPct = Math.max(0, 100 - pPct - fPct);


            return { pPct, fPct, cPct, pKcal, fKcal, cKcal, totalMacroKcal };

        }

        function round1(num) {

        function escapeHtml(str) {

            return String(str)

                .replace(/&/g, '&amp;')

                .replace(/</g, '&lt;')

                .replace(/>/g, '&gt;')

                .replace(/"/g, '&quot;')

                .replace(/'/g, '&#039;');

        }
